# Technical Information
help_technical_title = Technical Information

help_embeddings_title = About Embeddings & Semantic Search
help_embeddings_text = "Embedding support" means the model can be used for semantic search. A dedicated embedding model converts your documents into searchable vectors, then the chat model uses them to find the relevant information.
help_embeddings_with = <strong>With embedding support:</strong> cloud providers (OpenAI, Anthropic, Mistral, Gemini) handle both analytics and semantic search.
help_embeddings_without = <strong>Without embedding support:</strong> most local models (LM Studio, Ollama) are analytics only.

help_notes_title = Important Notes
help_notes_1 = Each model must be configured in the AI Models settings (API key, context window, pricing).
help_notes_2 = Local models require LM Studio or Ollama running on your server.
help_notes_3 = Response times vary with query complexity, the chosen model, and server load.
help_notes_4 = Analytics accuracy depends on the model: a reasoning model with a large context is the most reliable.

# LLM Configuration (local models)
help_llmconfig_title = Local LLM Configuration (LM Studio / Ollama)
help_llmconfig_intro = Local models must be able to hold ClicShopping's analytics prompt, which includes the database schema and can reach around 25,000 tokens. Configure the model accordingly on your local server.
help_llmconfig_context = <strong>Context window:</strong> load the model with a context of at least 32,000 tokens. A model limited to 16k (for example Phi-4) cannot answer analytics queries — the prompt alone exceeds its window and the reply comes back empty.
help_llmconfig_reasoning = <strong>Reasoning models</strong> (with an internal thinking step, such as Qwen3): they spend part of their output budget thinking before answering. Set a high enough output limit (LM Studio uses up to 8000 tokens) or the final answer is truncated or empty. Non-reasoning models avoid this and are faster.
help_llmconfig_speed = <strong>Response time:</strong> a local model can take several minutes per analytics answer depending on your hardware; cloud models answer in seconds.
help_llmconfig_reliability = Reasoning models with a large context window give the most reliable analysis.
help_llmconfig_pricing = Model usage pricing may change over time. Please keep it up to date.
